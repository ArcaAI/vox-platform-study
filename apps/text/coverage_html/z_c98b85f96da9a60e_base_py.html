21class CredentialPosture(StrEnum): 

22 """How an adapter obtains the credential it authenticates with. 

23 

24 This exists so the BYOK lock test can iterate the REGISTRY instead of a 

25 hand-written list of provider names. That distinction is not cosmetic: the 

26 previous list named azure/openai/anthropic, and underneath it ``bedrock`` 

27 built a ``boto3`` client with no credentials (silently authenticating from 

28 ``AWS_ACCESS_KEY_ID`` / ``AWS_PROFILE`` / instance metadata) and ``vertex`` 

29 built a ``genai`` client with no ``credentials=`` (Google ADC) — for years, 

30 because neither was on the list. A declaration each adapter must make, and 

31 a test that reads it off every registered provider, cannot skip the adapter 

32 nobody remembered. 

33 

34 An adapter that declares NOTHING fails the lock test. Default-deny is the 

35 point: a new provider is not silently assumed credential-free. 

36 """

51def require_model(model: str | None, *, provider: str) -> str: 

52 """Fail-closed guard for CLOUD providers (Azure/Bedrock/OpenAI/Anthropic/ 

53 Vertex): raise when no model resolved rather than falling through to a 

54 substituted default. 

55 

56 The five cloud sub-configs (``core/config.py``) carry no runtime 

57 fallback model — provider/model SELECTION is ``failMode=closed``, so an 

58 unresolved value must raise, never substitute a vendor model. Local 

59 built-in engines (Ollama, LM Studio/OpenAICompat, vLLM, llama.cpp) do 

60 NOT call this guard; they keep their topology-level model default. 

61 """ 

62 if not model or not model.strip(): 

63 raise ModelNotSelectedError( 

64 f"No model selected for provider '{provider}': Text does not " 

65 "substitute a default cloud model — the caller must supply " 

66 "'model' (resolved upstream by the gateway).", 

67 provider=provider, 

68 ) 

69 return model

90@runtime_checkable 

91class LLMProvider(Protocol): 

92 """Contract that every LLM provider must satisfy. 

93 

94 ``generate``/``generate_stream`` accept an OPTIONAL multimodal payload via 

95 ``request.content_parts`` — a text-only ``GenerateRequest`` (the 

96 default) is unaffected; a provider with no vision wire capability must 

97 raise ``VisionNotSupportedError`` (see ``reject_vision``) rather than drop 

98 the image parts. 

99 

100 Every implementation must also carry a ``credential_posture`` class attribute 

101 (see ``CredentialPosture``). It is deliberately NOT declared as a member of 

102 this Protocol: ``runtime_checkable`` turns any non-method member into an 

103 ``isinstance`` requirement, which would change ``isinstance(x, LLMProvider)`` 

104 for every duck-typed provider and test double in the codebase. The obligation 

105 is enforced instead by the registry-iterating lock test 

106 (``tests/unit/test_task602_byok_credentials.py``), which fails for any 

107 registered adapter that does not declare one. 

108 """

130@runtime_checkable 

131class ConnectionAwareProbe(Protocol): 

132 """An adapter that can enumerate an engine it is TOLD about. 

133 

134 ``get_info()`` on a self-hosted adapter probes ``self._last_base_url`` — the 

135 endpoint this process happened to serve a generation from last. That memo is 

136 process-wide, so it can only ever describe ONE engine per provider name, 

137 which is why model discovery could not show a tenant its own LM Studio or 

138 Ollama instance. 

139 

140 ``discover_models`` is the connection-scoped answer: the gateway resolves the 

141 caller's row through the one tenant → SYSTEM cascade and hands the endpoint 

142 down, exactly as ``/generate`` hands down ``provider_overrides``. The gateway 

143 still never opens an engine connection itself. 

144 

145 Two obligations for an implementation, both asserted by 

146 ``tests/unit/test_task799_provider_discovery.py``: 

147 

148 * build a REQUEST-SCOPED client and do NOT write ``_last_base_url`` — a 

149 probe describes an engine, it must never re-point the generation memo of a 

150 concurrently-serving process; 

151 * a connection with no ``api_key`` is the NORMAL self-hosted shape and must 

152 probe UNAUTHENTICATED, never raise. 

153 

154 Declared as a ``Protocol`` rather than added to ``LLMProvider`` because only 

155 the self-hosted engines can be enumerated at all: a cloud provider has no 

156 per-tenant model listing to fetch, and forcing every adapter to grow a method 

157 it cannot implement would buy nothing. ``_probe`` falls back to ``get_info()`` 

158 for anything that does not satisfy this. 

159 """

166class ProviderRegistry: 

167 """Registry of named LLMProviders with LAZY, request-driven instantiation. 

168 

169 a provider is *available* once its CONNECTION config is present 

170 (``register_factory`` at startup), but the (weight/connection-bearing) 

171 instance is built only on the first ``get(name)`` for that key and then 

172 memoized. A request for a name with neither an instance nor a factory 

173 (missing connection config) fails closed with ``ProviderNotFoundError``. 

174 Eager ``register`` is retained for tests / direct wiring. 

175 """ 

176 

177 def __init__(self) -> None: 

178 self._providers: dict[str, LLMProvider] = {} 

179 self._factories: dict[str, Callable[[], LLMProvider]] = {} 

180 

181 def register(self, name: str, provider: LLMProvider) -> None: 

182 """Register an already-built provider instance (eager).""" 

183 self._providers[name] = provider 

184 

185 def register_factory(self, name: str, factory: Callable[[], LLMProvider]) -> None: 

186 """Register a lazy builder; the instance is created on first ``get``.""" 

187 self._factories[name] = factory 

188 

189 def get(self, name: str) -> LLMProvider: 

190 provider = self._providers.get(name) 

191 if provider is not None: 

192 return provider 

193 factory = self._factories.get(name) 

194 if factory is not None: 

195 provider = factory() 

196 self._providers[name] = provider # memoize the lazily-built instance 

197 return provider 

198 raise ProviderNotFoundError( 

199 f"Provider '{name}' not registered. Available: {self.list_providers()}" 

200 ) from None 

201 

202 def unregister(self, name: str) -> None: 

203 self._providers.pop(name, None) 

204 self._factories.pop(name, None) 

205 

206 def list_providers(self) -> list[str]: 

207 """All AVAILABLE provider names (instantiated OR lazily registered).""" 

208 return list({**self._factories, **self._providers}.keys()) 

209 

210 def is_instantiated(self, name: str) -> bool: 

211 """True once the named provider has actually been built (test aid).""" 

212 return name in self._providers
