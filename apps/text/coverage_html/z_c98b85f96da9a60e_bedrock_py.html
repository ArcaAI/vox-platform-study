64class _StaticBearerTokenProvider: 

65 """A botocore token provider that always resolves to one fixed token. 

66 

67 A tenant's BYO Bedrock credential is a single ``api_key`` string — the 

68 AWS "Bedrock API key" bearer-token auth, not a SigV4 access/secret key 

69 pair — so overriding it means handing botocore a token *provider*, not a 

70 static credential. Scoped to a dedicated ``botocore.session.Session`` 

71 built fresh per request (never the process-wide default session or an 

72 env var), so two tenants' overrides on concurrent requests can never 

73 race each other. 

74 """ 

75

77 self._token = token

83# ── Network bytes without a transport to hook (TASK-959 §4.1) ──────────────── 

84# 

85# Every other adapter in this service reaches its vendor through 

86# `pooled_http_client`, so `providers/pool.py`'s transport counts its bytes. 

87# Bedrock is botocore, not httpx, and it never will be on that pool — so it gets 

88# the equivalent from botocore's OWN event system, credited into the same 

89# per-request record the pooled transport uses. 

90# 

91# `before-send` carries the prepared request, whose `body` is the serialized 

92# JSON. `response-received` carries a response dict whose `body` is the response 

93# BYTES for a normal operation — and, for an operation with a streaming output 

94# shape (`converse_stream`), the UNREAD event stream. Reading that to size it 

95# would consume the generation, so it is not read: the request is counted, the 

96# response is not, and `usage_detail` answers `response_bytes: None` rather than 

97# a zero that would read as "the vendor sent nothing". That is §4.2's documented 

98# blind spot, stated where it is created. 

99# 

100# The hooks are registered on the CLIENT, which is cached and shared by every 

101# request on the same (region, credential) — so registration happens once, in the 

102# factory, and attribution comes from the contextvar, which `with_byte_counts` 

103# carries into the Bedrock worker thread the SDK actually runs on. 

104 

105#: Botocore event prefixes, wildcard over every operation of the service: one 

106#: registration covers `Converse` and `ConverseStream` and anything added later. 

107_BEFORE_SEND_EVENT = "before-send.bedrock-runtime" 

108_RESPONSE_RECEIVED_EVENT = "response-received.bedrock-runtime"

111def _count_request_bytes(request: Any = None, **_kwargs: Any) -> None: 

112 """Count the prepared request's body. MUST return ``None``. 

113 

114 A `before-send` handler that returns anything else REPLACES the HTTP call 

115 with its return value (that is how botocore's stubber works), so a metering 

116 hook that accidentally returned a truthy value would silently stop every 

117 Bedrock generation from reaching AWS. 

118 """ 

119 counts = current_byte_counts() 

120 if counts is None: 

121 return 

122 body = getattr(request, "body", None) 

123 counts.add_request(len(body) if isinstance(body, (bytes, bytearray)) else 0) 

124 

125 

126def _count_response_bytes(response_dict: Any = None, **_kwargs: Any) -> None: 

127 """Count the response body when it is bytes; never consume a stream.""" 

128 counts = current_byte_counts() 

129 if counts is None or not isinstance(response_dict, dict): 

130 return 

131 body = response_dict.get("body") 

132 if isinstance(body, (bytes, bytearray)): 

133 counts.add_response(len(body)) 

134 return 

135 # A streaming body. `Content-Length` is absent on a chunked event stream, so 

136 # this is the non-streaming-but-not-buffered case, not a second guess at the 

137 # stream's size. 

138 headers = response_dict.get("headers") or {} 

139 declared = headers.get("content-length") if hasattr(headers, "get") else None 

140 try: 

141 if declared is not None: 

142 counts.add_response(max(0, int(declared))) 

143 except (TypeError, ValueError): 

144 return

147def _register_byte_hooks(client: Any) -> Any: 

148 """Attach the byte hooks to one freshly built client, and return it.""" 

149 client.meta.events.register(_BEFORE_SEND_EVENT, _count_request_bytes) 

150 client.meta.events.register(_RESPONSE_RECEIVED_EVENT, _count_response_bytes) 

151 return client

154def _bearer_client( 

155 service: Literal["bedrock-runtime", "bedrock"], *, token: str, region: str 

156) -> Any: 

157 """A boto3 client bound to ONE explicit bearer token. 

158 

159 Every Bedrock client in this module is built here, platform and tenant alike, 

160 so there is exactly one construction site and it always takes a credential. 

161 The session is dedicated (never the process-wide default session), so two 

162 tenants' concurrent requests can never race each other's token. 

163 """ 

164 session = botocore.session.Session() 

165 # botocore-stubs doesn't type this private attribute. 

166 session._components.register_component( # type: ignore[attr-defined] 

167 "token_provider", _StaticBearerTokenProvider(token) 

168 ) 

169 return _register_byte_hooks( 

170 boto3.Session(botocore_session=session).client( 

171 service, 

172 region_name=region, 

173 config=BotocoreConfig(signature_version="bearer"), 

174 ) 

175 )

183 # TASK-970 — UNSUPPORTED, and that is a property of the CONVERSE API, not an 

184 # oversight. Verified against the pinned service model (botocore==1.43.72, 

185 # `bedrock-runtime/2023-09-30`): `ConverseRequest` has no reasoning member at 

186 # all. The only route is `additionalModelRequestFields`, a free-form 

187 # `Document` passed straight through to the UNDERLYING model's own 

188 # vocabulary — which differs by family and generation (`reasoning_config` on 

189 # Claude 3.7, `thinking` on Sonnet 4 / adaptive on Opus), and where a key the 

190 # family does not know is a ValidationException, i.e. a FAILED generation. 

191 # A normalized posture cannot be rendered safely here, so it is recorded and 

192 # the call proceeds — the owner decision, and the opposite of today's silent 

193 # drop. Enabling it needs the bound model to declare its own family 

194 # vocabulary first. 

195 reasoning_support = ReasoningSupport.UNSUPPORTED 

196 reasoning_parameter = None 

197 reasoning_effort_parameter = None

199 def __init__(self) -> None: 

200 """No configuration, and NO AMBIENT CREDENTIAL CHAIN. 

201 

202 The credential AND the region both arrive per request as a 

203 gateway-resolved ``ProviderOverride``. There is no shared client, so 

204 every request builds its own bearer-token client and two tenants can 

205 never race on one. 

206 

207 This adapter used to hold a bare `boto3.client("bedrock-runtime", ...)`, 

208 which resolves credentials from `AWS_ACCESS_KEY_ID` / `AWS_PROFILE` / EC2 

209 instance metadata — a working PLATFORM identity that appeared in no env 

210 file, no `turbo.json` and no registry, on which every tenant's traffic 

211 silently ran. Removing the config field is what makes that unreachable: 

212 there is no longer any path to a client built without an explicit token. 

213 """

226 def _client_for(self, request: GenerateRequest) -> Any: 

227 """Request-scoped, fail-closed bearer-token client.""" 

228 override = self._resolve_override(request) 

229 if override is None: 

230 raise ProviderConnectionMissingError( 

231 "No AWS Bedrock connection resolved. Bedrock is BYOK-only: " 

232 "configure a tenant Bedrock credential, or the platform " 

233 "(SYSTEM-tenant) connection, in the provider-connection plane. " 

234 "There is no env fallback and no ambient AWS credential chain.", 

235 provider="bedrock", 

236 ) 

237 region = (override.region or "").strip() 

238 if not region: 

239 raise ProviderConnectionMissingError( 

240 "No AWS region on the resolved Bedrock connection. A Bedrock " 

241 "client is bound to a region, so it travels with the credential " 

242 "on the AiProviderConnection row.", 

243 provider="bedrock", 

244 ) 

245 token = override.api_key.get_secret_value() 

246 # Region is part of the identity: a boto3 client is bound to one, so the 

247 # same credential in two regions is two clients. 

248 key = client_key("bedrock", region, token) 

249 return CLIENT_CACHE.get_or_create( 

250 key, 

251 lambda: _bearer_client("bedrock-runtime", token=token, region=region), 

252 )

255 resolved = resolve_request_defaults(request) 

256 content: list[dict[str, Any]] = [{"text": request.prompt}] 

257 for image in request.image_parts(): 

258 content.append(_bedrock_image_block(image)) 

259 params: dict[str, Any] = { 

260 "modelId": self._resolve_model(request), 

261 "messages": [{"role": "user", "content": content}], 

262 "inferenceConfig": { 

263 "temperature": resolved["temperature"], 

264 "maxTokens": resolved["max_tokens"], 

265 "topP": resolved["top_p"], 

266 }, 

267 } 

268 if request.system_prompt: 

269 params["system"] = [{"text": request.system_prompt}] 

270 

271 if request.response_format is not None and request.response_format.type == "json_schema": 

272 schema = request.response_format.json_schema or {} 

273 params["toolConfig"] = { 

274 "tools": [ 

275 { 

276 "toolSpec": { 

277 "name": schema.get("title", "output"), 

278 "description": "Structured output schema", 

279 "inputSchema": {"json": schema}, 

280 } 

281 } 

282 ], 

283 "toolChoice": {"tool": {"name": schema.get("title", "output")}}, 

284 }

301 async def generate(self, request: GenerateRequest) -> tuple[str, str, GenerationStats]: 

302 resolved = resolve_request_defaults(request) 

303 resolved_model = require_model(self._resolve_model(request), provider="bedrock") 

304 with _get_tracer().start_as_current_span( 

305 "gen_ai.generate", 

306 attributes={ 

307 "gen_ai.system": "aws_bedrock", 

308 "gen_ai.request.model": resolved_model, 

309 "gen_ai.operation.name": "generate", 

310 "gen_ai.request.temperature": resolved["temperature"], 

311 "gen_ai.request.max_tokens": resolved["max_tokens"], 

312 }, 

313 ) as span: 

314 params = self._build_converse_params(request) 

315 client = self._client_for(request) 

316 start = time.monotonic() 

317 # Same boundary as the streaming path below, for the same reason: 

318 # `asyncio.to_thread` runs on the DEFAULT executor, and a 

319 # non-streaming Bedrock generation holds its thread for the whole 

320 # generation — seconds, not milliseconds. Bedrock's blocking work 

321 # draws on Bedrock's own budget or it draws on everyone's. 

322 response = await run_in_bedrock_executor(lambda: client.converse(**params)) 

323 total_ms = int((time.monotonic() - start) * 1000) 

324 

325 stop_reason = response.get("stopReason", "") 

326 if stop_reason == "guardrail_intervened": 

327 logger.warning( 

328 "bedrock.guardrail_intervened", 

329 model=params["modelId"], 

330 guardrail_id=params.get("guardrailConfig", {}).get("guardrailIdentifier"), 

331 ) 

332 

333 content_blocks = response["output"]["message"]["content"] 

334 content = "".join(block.get("text", "") for block in content_blocks) 

335 # ReasoningContentBlockOutputTypeDef nests the text under "reasoningText" 

336 # (unlike the streaming delta variant, which has a flat "text" field). 

337 reasoning = "".join( 

338 block.get("reasoningContent", {}).get("reasoningText", {}).get("text", "") 

339 for block in content_blocks 

340 ) 

341 

342 raw_usage: dict[str, Any] = dict(response.get("usage", {})) 

343 stats = stats_from_bedrock( 

344 provider="bedrock", 

345 model=resolved_model, 

346 usage=raw_usage, 

347 stop_reason=stop_reason, 

348 total_ms=total_ms, 

349 engine_native={"usage": raw_usage, "stopReason": stop_reason}, 

350 ) 

351 span.set_attribute("gen_ai.usage.input_tokens", stats.prompt_tokens) 

352 span.set_attribute("gen_ai.usage.output_tokens", stats.predicted_tokens) 

353 span.set_attribute("gen_ai.response.finish_reason", stop_reason or "stop") 

354 return content, reasoning, stats

356 async def generate_stream(self, request: GenerateRequest) -> AsyncIterator[StreamChunk]: 

357 resolved = resolve_request_defaults(request) 

358 resolved_model = require_model(self._resolve_model(request), provider="bedrock") 

359 with _get_tracer().start_as_current_span( 

360 "gen_ai.generate_stream", 

361 attributes={ 

362 "gen_ai.system": "aws_bedrock", 

363 "gen_ai.request.model": resolved_model, 

364 "gen_ai.operation.name": "generate_stream", 

365 "gen_ai.request.temperature": resolved["temperature"], 

366 "gen_ai.request.max_tokens": resolved["max_tokens"], 

367 }, 

368 ) as span: 

369 params = self._build_converse_params(request) 

370 client = self._client_for(request) 

371 

372 loop = asyncio.get_running_loop() 

373 queue: asyncio.Queue[Any] = asyncio.Queue() 

374 _SENTINEL = object() 

375 

376 def _iterate_stream() -> None: 

377 """Run in thread — iterates boto3 sync stream, pushes to queue.""" 

378 try: 

379 response = client.converse_stream(**params) 

380 for event in response["stream"]: 

381 loop.call_soon_threadsafe(queue.put_nowait, event) 

382 loop.call_soon_threadsafe(queue.put_nowait, _SENTINEL) 

383 except Exception as exc: 

384 loop.call_soon_threadsafe(queue.put_nowait, exc) 

385 

386 # NOT the default executor (B-3). `_iterate_stream` holds its thread 

387 # for the ENTIRE stream, not for one call, so on asyncio's default 

388 # pool (`min(32, cpu + 4)`) roughly 32 concurrent Bedrock generations 

389 # stall every other `asyncio.to_thread` caller in the process — a 

390 # ceiling Bedrock imposes on everybody else. The dedicated pool is 

391 # sized from the same control-plane `maxConcurrent` that bounds this 

392 # provider's semaphore, so it is never the binding constraint and 

393 # never anyone else's problem. See `providers/pool.py`. 

394 thread_future = loop.run_in_executor( 

395 bedrock_stream_executor(), 

396 # TASK-959 — the botocore hooks fire on THIS thread, and 

397 # `run_in_executor` carries no context into it. 

398 with_byte_counts(_iterate_stream), 

399 ) 

400 

401 # DRAIN to completion (AD-1): the ``messageStop`` (stopReason) event 

402 # arrives BEFORE the ``metadata`` (usage) event, so yielding ``done`` 

403 # at ``messageStop`` emits done→usage (wrong order) and drops usage 

404 # from the stats. Accumulate stop reason + usage across the whole 

405 # stream, measure TTFT at the first content/reasoning delta, then emit 

406 # ONE ``usage`` chunk (full AD-1 stats) followed by ``done``. 

407 start = time.monotonic() 

408 ttft_ms: int | None = None 

409 stop_reason: str | None = None 

410 raw_usage: dict[str, Any] = {} 

411 try: 

412 while True: 

413 event = await queue.get() 

414 if event is _SENTINEL: 

415 break 

416 if isinstance(event, Exception): 

417 raise event 

418 

419 if "contentBlockDelta" in event: 

420 delta = event["contentBlockDelta"]["delta"] 

421 reasoning_text = delta.get("reasoningContent", {}).get("text", "") 

422 if reasoning_text: 

423 if ttft_ms is None: 

424 ttft_ms = int((time.monotonic() - start) * 1000) 

425 yield StreamChunk(type="reasoning", content=reasoning_text) 

426 text = delta.get("text", "") 

427 if text: 

428 if ttft_ms is None: 

429 ttft_ms = int((time.monotonic() - start) * 1000) 

430 yield StreamChunk(type="chunk", content=text) 

431 elif "messageStop" in event: 

432 stop_reason = event["messageStop"].get("stopReason", stop_reason) 

433 if stop_reason == "guardrail_intervened": 

434 # Parity with `generate()`. A guardrail intervention 

435 # reaches the streaming path through the very same 

436 # `stopReason` field, and it is a safety event: it 

437 # must be as visible in the logs for a streamed 

438 # generation as for a synchronous one. 

439 logger.warning( 

440 "bedrock.guardrail_intervened", 

441 model=params["modelId"], 

442 guardrail_id=params.get("guardrailConfig", {}).get( 

443 "guardrailIdentifier" 

444 ), 

445 ) 

446 elif "metadata" in event: 

447 raw_usage = event["metadata"].get("usage", {}) or {} 

448 finally: 

449 await thread_future 

450 

451 total_ms = int((time.monotonic() - start) * 1000) 

452 stats = stats_from_bedrock( 

453 provider="bedrock", 

454 model=resolved_model, 

455 usage=raw_usage, 

456 stop_reason=stop_reason, 

457 total_ms=total_ms, 

458 ttft_ms=ttft_ms, 

459 engine_native=( 

460 {"usage": raw_usage, "stopReason": stop_reason} 

461 if raw_usage or stop_reason 

462 else None 

463 ), 

464 ) 

465 if raw_usage: 

466 span.set_attribute("gen_ai.usage.input_tokens", raw_usage.get("inputTokens", 0)) 

467 span.set_attribute("gen_ai.usage.output_tokens", raw_usage.get("outputTokens", 0)) 

468 span.set_attribute("gen_ai.response.finish_reason", stop_reason or "stop") 

469 yield StreamChunk(type="usage", data=stats.model_dump()) 

470 # The terminal frame reports the reason the STATS beside it report. 

471 # ``MessageStopEvent.stopReason`` is REQUIRED by the ConverseStream 

472 # contract, so a stream that ended without a ``messageStop`` did not 

473 # finish — substituting the literal ``"stop"`` here billed and 

474 # audited a truncated generation as a complete one, while the very 

475 # same frame's stats said ``"other"``. The two must never disagree. 

476 yield StreamChunk(type="done", data={"finish_reason": stop_reason or stats.stop_reason})
