117# --------------------------------------------------------------------------- 

118# Network consumption per upstream (TASK-959 M-4) 

119# --------------------------------------------------------------------------- 

120# Counted at the ONE seam every pooled adapter already passes through — the 

121# draining transport in `providers/pool.py` — so a new adapter on the pool is 

122# metered by construction rather than by remembering to instrument it. 

123# 

124# `direction` uses the ledger's own vocabulary (`EGRESS_BYTE` / `INGRESS_BYTE`) 

125# rather than request/response, because these counters and the ledger rows they 

126# corroborate must be comparable without a translation table. `funding` is the 

127# tier that paid for the call (`tenant` = BYOK, `platform` = the platform's own 

128# credential), so a byte total can be split the same way spend already is. 

129# 

130# Bodies only: request and response BODY bytes, never headers or TLS framing. 

131# Stated on the metric because a capacity plan built on it would otherwise 

132# silently under-count by the header budget of every call. 

133 

134PROVIDER_BYTES = Counter( 

135 "text_provider_bytes_total", 

136 "Body bytes exchanged with an upstream provider, by direction and funding tier", 

137 ["provider", "direction", "funding"], 

138)

189@contextmanager 

190def track_model_inference(model: str, service: str = SERVICE_NAME) -> Iterator[None]: 

191 """Track one model inference: bump the running gauge for its duration and 

192 observe its latency. The gauge is always decremented, even on error. 

193 """ 

194 MODEL_RUNNING_INSTANCES.labels(service=service, model=model).inc() 

195 start = time.perf_counter() 

196 try: 

197 yield 

198 finally: 

199 MODEL_INFERENCE_LATENCY.labels(service=service, model=model).observe( 

200 time.perf_counter() - start 

201 ) 

202 MODEL_RUNNING_INSTANCES.labels(service=service, model=model).dec()
