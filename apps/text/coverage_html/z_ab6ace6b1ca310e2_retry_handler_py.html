1"""Retry logic with exponential backoff for provider errors. 

2 

3Policy (Lane C C-6): **at most 3 retries, exponential backoff, 

4always jittered — including when the upstream handed us an exact wait — and 

5`Retry-After` is authoritative when the upstream sent one.** 

6 

7The "always jitter, even with an exact wait" clause is the non-obvious half. An 

8exact `Retry-After` is precisely the case where every rate-limited caller has 

9been given the SAME deadline by the SAME upstream, so obeying it to the 

10millisecond synchronises the herd instead of spreading it. 

11 

12TASK-946 D5: a provider 4xx is a DETERMINISTIC rejection of this exact 

13request (a malformed body, an oversized prompt) — retrying it burns the 

14platform's retry budget against an outcome that cannot change, and the 

15generic 502 that used to follow reads as "the provider is down" rather than 

16"the request was rejected". `provider_status_code_from` / `provider_error_code_from` 

17below classify that shape; `should_retry` refuses it unconditionally. 

18 

19TASK-993 D-5: that sweep was written for a malformed body and took **429 and 

20408 with it**, which are the opposite kind of 4xx — the provider answered 

21promptly and correctly, and (for a 429) named a deadline it is asking us to 

22honour. Retrying them is not waste, it is the whole point, so 

23`_TRANSIENT_PROVIDER_STATUSES` carves them back out. A 429 additionally gets 

24its own `RATE_LIMITED_ERROR_TYPE`, which `should_retry` admits STRUCTURALLY — 

25the exact mirror of the `invalid_request` refusal, for the mirror reason: it is 

26the one error the upstream has explicitly asked us to retry, and a caller that 

27skips the backoff amplifies the incident for everyone sharing that upstream. 

28"""

36#: Hard ceiling on retries, whatever a caller's ``RetryConfig`` asks for. A 

37#: request body is not allowed to raise the platform's amplification factor 

38#: against a struggling provider. 

39MAX_RETRIES = 3 

40 

41#: Jitter as a fraction of the delay. Kept at the long-standing 10% rather than 

42#: widened to full jitter: the existing value is already load-bearing (see 

43#: `test_service_edge_cases.py`) and nothing measured justifies changing it. 

44_JITTER_FRACTION = 0.1

47def calculate_backoff( 

48 attempt: int, 

49 base_delay: float = 1.0, 

50 max_delay: float = 60.0, 

51 *, 

52 retry_after: float | None = None, 

53) -> float: 

54 """Seconds to wait before the next attempt. 

55 

56 ``retry_after`` — the wait an upstream explicitly asked for — wins over the 

57 computed exponential delay, but is still capped by ``max_delay`` (an upstream 

58 is allowed to ask for an hour; we are not obliged to hold a request that 

59 long) and still jittered, per the module docstring. 

60 """ 

61 if retry_after is not None and retry_after > 0: 

62 delay = min(retry_after, max_delay) 

63 else: 

64 delay = min(base_delay * (2**attempt), max_delay) 

65 jitter = random.uniform(0.0, delay * _JITTER_FRACTION) 

66 return min(delay + jitter, max_delay)

69def retry_after_from(exc: BaseException) -> float | None: 

70 """The wait, in seconds, that ``exc`` reports its upstream asked for. 

71 

72 Two shapes, because two kinds of error reach the retry loop: 

73 

74 * our own typed errors, which carry a ``retry_after`` attribute already; and 

75 * a provider SDK's HTTP error, which carries the raw response — where 

76 Retry-After is either delta-seconds or an HTTP-date (RFC 9110 §10.2.3). 

77 Both forms are accepted; anything else yields ``None``. 

78 

79 ``None`` means "no opinion", never "zero" — a malformed header must fall back 

80 to the computed exponential backoff rather than retrying immediately. 

81 """ 

82 direct = getattr(exc, "retry_after", None) 

83 if isinstance(direct, int | float) and direct > 0: 

84 return float(direct) 

85 

86 response = getattr(exc, "response", None) 

87 headers = getattr(response, "headers", None) 

88 if headers is None: 

89 return None 

90 try: 

91 raw = headers.get("retry-after") or headers.get("Retry-After") 

92 except (AttributeError, TypeError): 

93 return None 

94 if not raw: 

95 return None 

96 

97 raw = str(raw).strip() 

98 try: 

99 seconds = float(raw) 

100 except ValueError: 

101 try: 

102 when = parsedate_to_datetime(raw) 

103 except (TypeError, ValueError): 

104 return None 

105 if when.tzinfo is None: 

106 when = when.replace(tzinfo=UTC) 

107 seconds = (when - datetime.now(UTC)).total_seconds() 

108 return seconds if seconds > 0 else None

111#: The error-type value a deterministic provider 4xx is classified as (see 

112#: `provider_status_code_from` below). Structurally excluded from 

113#: `should_retry` — never retried, whatever a caller's ``retry_on`` asks for. 

114INVALID_REQUEST_ERROR_TYPE = "invalid_request" 

115 

116#: The `code` this service answers with for a provider 4xx that is NOT 

117#: recognised as a context-window overflow. Same naming convention as 

118#: `services/output_gate.py`'s `REJECTED_CODE` / `UNAVAILABLE_CODE`. 

119PROVIDER_INVALID_REQUEST_CODE = "PROVIDER_INVALID_REQUEST" 

120 

121#: The `code` for a provider 4xx whose message says the prompt (plus history/ 

122#: system prompt) no longer fits the model's context window — the D5 trial 

123#: evidence (LM Studio: ``"Context size has been exceeded."``). Checked 

124#: case-insensitively as a substring, not an exact match: every engine phrases 

125#: this differently and none of them hand back a machine-readable field for 

126#: it, only prose. 

127CONTEXT_WINDOW_EXCEEDED_CODE = "CONTEXT_WINDOW_EXCEEDED" 

128 

129_CONTEXT_WINDOW_PHRASES: tuple[str, ...] = ( 

130 "context size", 

131 "context length", 

132 "context window", 

133 "maximum context", 

134 "too many tokens", 

135 "exceeds the model", 

136) 

137 

138#: The 4xx statuses that are TRANSIENT rather than deterministic — the provider 

139#: rejected this ATTEMPT, not this REQUEST, so a later one can succeed. 429 

140#: ("you are going too fast") and 408 ("that attempt took too long") are the 

141#: only two: every other 4xx describes the request itself. 

142_TRANSIENT_PROVIDER_STATUSES = frozenset({408, 429}) 

143 

144#: The error-type value a vendor 429 is classified as. Admitted by 

145#: `should_retry` structurally, bounded only by the attempt ceiling. 

146RATE_LIMITED_ERROR_TYPE = "rate_limited" 

147 

148#: The `code` both surfaces answer with for a vendor 429 — the blocking route's 

149#: `error_code` (via `RateLimitError`) and the streaming producer's terminal 

150#: frame. ONE token, because a client should not have to learn two names for 

151#: one condition; `test_retry_and_timeout.py` pins the two together. 

152RATE_LIMITED_CODE = "RATE_LIMITED" 

153 

154#: The wait advertised to the CALLER when a vendor 429 exhausted the retry 

155#: budget and the vendor itself named no deadline. A 429 with no `Retry-After` 

156#: is a 429 nothing can pace against, which invites the same storm straight 

157#: back. Not config and not a tuning knob: it is this error type's contract, 

158#: exactly like `CircuitOpenError`'s 30s and `ConcurrencyLimitError`'s 5s in 

159#: `core/exception_handlers.py`. 

160DEFAULT_RATE_LIMIT_RETRY_AFTER_S = 5.0

163def provider_status_code_from(exc: BaseException) -> int | None: 

164 """The HTTP status code a provider's own response carried, if any. 

165 

166 Duck-typed across the two shapes Text's provider adapters actually raise 

167 (verified against the installed SDKs — see the D5 evidence in 

168 ``api/endpoints/generate.py``, not guessed): 

169 

170 * ``openai.APIStatusError`` (LM Studio, vLLM, the generic OpenAI-compat 

171 adapter, Azure OpenAI) and ``anthropic.APIStatusError`` both expose 

172 ``.status_code`` directly; and 

173 * a raw ``httpx.HTTPStatusError`` (Ollama, llama.cpp, TEI — 

174 ``resp.raise_for_status()``) carries it on ``.response.status_code`` 

175 instead. 

176 

177 Neither shape is imported here, so a provider adapter added later needs no 

178 update to this function as long as it raises one of these two familiar 

179 shapes. A Bedrock ``botocore.exceptions.ClientError`` matches neither 

180 (its ``.response`` is a plain ``dict``, not an ``httpx.Response``) and 

181 correctly falls through to ``None`` — unclassified, exactly like before 

182 this function existed. 

183 """ 

184 direct = getattr(exc, "status_code", None) 

185 if isinstance(direct, int): 

186 return direct 

187 response = getattr(exc, "response", None) 

188 status = getattr(response, "status_code", None) 

189 return status if isinstance(status, int) else None

192def is_provider_rate_limited(exc: BaseException) -> bool: 

193 """Whether ``exc`` is a vendor 429 — the provider throttling us. 

194 

195 Deliberately narrower than "transient 4xx": a 408 is retryable too, but it 

196 is not a rate limit, so it must not inherit the 429 RESPONSE mapping. A 

197 caller told "429, Retry-After" pauses; a caller told 429 for something 

198 nobody rate-limited pauses for no reason. 

199 """ 

200 return provider_status_code_from(exc) == 429

203def is_provider_invalid_request(exc: BaseException) -> bool: 

204 """Whether ``exc`` is a deterministic provider 4xx — never retryable. 

205 

206 The transient statuses are excluded (TASK-993 D-5): they are 4xx by status 

207 only, and refusing to retry them is the defect, not the policy. 

208 """ 

209 status_code = provider_status_code_from(exc) 

210 if status_code is None or not (400 <= status_code < 500): 

211 return False 

212 return status_code not in _TRANSIENT_PROVIDER_STATUSES

215def provider_error_code_from(exc: BaseException) -> str: 

216 """The response ``code`` for a provider 4xx already confirmed via 

217 :func:`is_provider_invalid_request`. 

218 

219 ``str(exc)`` is used rather than a specific SDK's ``.message`` attribute so 

220 an unrecognised exception type still classifies correctly instead of 

221 raising — the same "duck-type, don't import every SDK" posture as 

222 :func:`provider_status_code_from`. 

223 """ 

224 text = str(exc).lower() 

225 if any(phrase in text for phrase in _CONTEXT_WINDOW_PHRASES): 

226 return CONTEXT_WINDOW_EXCEEDED_CODE 

227 return PROVIDER_INVALID_REQUEST_CODE

230def should_retry( 

231 error_type: str, 

232 retry_on: list[str], 

233 attempt: int = 0, 

234 max_retries: int = 3, 

235) -> bool: 

236 """Determine if the request should be retried. 

237 

238 ``attempt`` is 0-based, so ``MAX_RETRIES = 3`` permits attempts 0, 1 and 2 to 

239 answer ``True`` — three retries on top of the original call. A caller asking 

240 for more is silently held to the ceiling rather than refused: the request is 

241 still serviceable, it just does not get to choose the platform's retry 

242 budget. 

243 

244 A deterministic provider 4xx (``INVALID_REQUEST_ERROR_TYPE``) is refused 

245 UNCONDITIONALLY — before the attempt-ceiling check and regardless of 

246 ``retry_on`` — because no number of retries changes an outcome the 

247 provider has already decided synchronously. This is structural, not a 

248 ``retry_on`` default: a caller that explicitly lists ``"invalid_request"`` 

249 still gets exactly one attempt. 

250 

251 A vendor 429 (``RATE_LIMITED_ERROR_TYPE``) is the exact mirror: ADMITTED 

252 structurally, whatever ``retry_on`` says, because it is the one error the 

253 upstream has explicitly asked us to retry — and a request that skips the 

254 backoff does not just fail itself, it keeps the upstream saturated for 

255 every other tenant on it. ``max_retries`` still bounds it, so a caller that 

256 genuinely wants one attempt asks for one attempt; ``retry_on`` is a 

257 which-error-CLASSES knob and was never the right place to disable the one 

258 class that comes with a deadline attached. 

259 """ 

260 if error_type == INVALID_REQUEST_ERROR_TYPE: 

261 return False 

262 if attempt >= min(max_retries, MAX_RETRIES): 

263 return False 

264 if error_type == RATE_LIMITED_ERROR_TYPE: 

265 return True 

266 return error_type in retry_on
