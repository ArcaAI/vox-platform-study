1"""Generation hyperparameter defaults — applied only when a request omits them. 

2 

3These three numbers (`temperature`, `max_tokens`, `top_p`) had NO config surface 

4at all: they were literals in this module, applied at `openai_compat.py`, 

5`ollama.py` and `bedrock.py`, and there was no way to change them short of a 

6redeploy. They are also the most obviously per-deployment values in the service — 

7a summarisation profile and a clinical-extraction profile want different ones. 

8 

9They have a home now. The gateway's ``applyTextRuntimeProfile`` 

10(`packages/applications/.../text-request-enrichment.service.ts`) already fills 

11`temperature` / `top_p` / `max_tokens` from `AiRuntimeProfile` onto the request 

12body when it can resolve one, so a request that carried a profile never reaches 

13the fallback below. What is added here is the case it CANNOT cover: a caller with 

14no resolvable `(provider, model)` profile, whose values now come from the 

15platform generation profile on the PULL channel instead of from a literal. 

16 

17Resolution order, most specific first: 

18 

191. the request's own value (a caller that set it always wins), 

202. the platform generation profile from `/internal/effective-config`, 

213. the in-code floor — a conservative, low-variance setting that is safe to send 

22 to any engine, not a preference anyone expressed. 

23 

24The active platform values live in module state rather than being threaded 

25through every adapter signature, mirroring how `apply_provider_retention` 

26(`services/runtime_limits.py`) pushes the retention TTL into live providers: one 

27refresh updates the process, and an adapter reads whatever is current. 

28""" 

29 

30from __future__ import annotations 

31 

32from typing import TYPE_CHECKING, Any 

33 

34if TYPE_CHECKING: 

35 from text.models.requests import GenerateRequest 

36 

37#: Last-resort floors. Deliberately conservative: near-deterministic sampling and 

38#: a generous-but-bounded output ceiling, chosen so that sending them to an engine 

39#: nobody profiled cannot produce a surprising bill or a truncated clinical note. 

40GENERATION_FLOOR: dict[str, float | int] = { 

41 "temperature": 0.1, 

42 "max_tokens": 16_384, 

43 "top_p": 0.95, 

44} 

45 

46#: The platform generation profile currently served by the control plane. 

47#: Replaced wholesale by `apply_generation_defaults`; empty until the first 

48#: successful effective-config refresh, and left untouched when the gateway is 

49#: unreachable (the negative-cache path) so a down control plane behaves exactly 

50#: like a control plane with no opinion. 

51_ACTIVE: dict[str, float | int] = {} 

52 

53 

54def apply_generation_defaults(served: dict[str, float | int]) -> None: 

55 """Adopt the platform generation profile. Called on every config refresh.""" 

56 _ACTIVE.clear() 

57 _ACTIVE.update(served) 

58 

59 

60def active_generation_defaults() -> dict[str, float | int]: 

61 """The effective defaults: control-plane values over the in-code floors.""" 

62 return {**GENERATION_FLOOR, **_ACTIVE} 

63 

64 

65def resolve_request_defaults(request: GenerateRequest) -> dict[str, float | int]: 

66 """Return resolved hyperparameters, using defaults only for None values.""" 

67 defaults = active_generation_defaults() 

68 values: dict[str, Any] = { 

69 "temperature": request.temperature, 

70 "max_tokens": request.max_tokens, 

71 "top_p": request.top_p, 

72 } 

73 return {name: defaults[name] if value is None else value for name, value in values.items()}
