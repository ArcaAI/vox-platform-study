1"""The keyed SDK-client cache — B-2, and the isolation property it must not cost. 

2 

3Every adapter used to build a fresh SDK client on EVERY request, and every one of 

4them carried a comment explaining why: *"there is no shared client, so two 

5tenants can never race on one."* That reasoning is correct and it is not 

6negotiable. A client carries a credential; a client two tenants share is a 

7credential two tenants share, and on the cloud adapters that is un-invoiced spend 

8plus a silent cross-tier substitution — the exact failure lane B closed. 

9 

10But "a new client per request" is a much stronger statement than the isolation 

11property needs, and at the concurrency this service is being built for it is 

12expensive: a TLS handshake per stream, and no connection reuse anywhere. 

13 

14The precise property is: 

15 

16 two requests may share a client **iff** they would authenticate identically 

17 to the same endpoint. 

18 

19So the cache key is exactly that predicate, and nothing else: 

20 

21 (provider, base_url, credential fingerprint) 

22 

23## The fingerprint is a salted hash, never the credential 

24 

25Keying on the credential itself would put every tenant's key in a long-lived 

26process-wide dict — a credential store nobody designed, reachable from any code 

27that can reach the cache. Keying on a BARE hash is barely better: an API key has 

28low enough entropy in places (and a known prefix everywhere) that a bare digest 

29is a dictionary attack away from the key. 

30 

31A per-process random salt fixes both. The digest is meaningless outside this 

32process and unrecoverable inside it, which is all a cache key ever needs to be: 

33equal for equal credentials, different for different ones. 

34 

35## Bounded, expiring, evictable 

36 

37* **Bounded** (LRU) so a tenant churn cannot grow the map without limit. 

38* **Expiring** so a client for a connection that no longer exists eventually 

39 goes away even if nothing tells us. 

40* **Evictable** on `arca:config:invalidate`, which is the real propagation path 

41 (rule 09 "Config caches": invalidation propagates, the TTL is a 

42 bounded-staleness net). A rotated credential already produces a NEW key and so 

43 can never be served from the old entry — eviction is about not leaving a client 

44 authenticated with a revoked key holding warm sockets. 

45 

46Eviction DROPS the reference; it does not close the client. A request may still 

47be streaming through it, and closing a live pool would abort a generation. The 

48sockets close when the last reference goes. 

49"""

71#: Per-process salt. Regenerated on every start, so a fingerprint is meaningless 

72#: to anything outside this process — including a heap dump taken later. 

73_CACHE_KEY_SALT = secrets.token_bytes(32) 

74 

75#: The key a connection with NO credential gets. A keyless self-hosted row (LM 

76#: Studio, vLLM, Ollama behind a topology `base_url`) is a distinct STATE, not a 

77#: credential, and giving it its own marker keeps it from colliding with a 

78#: hypothetical empty-string key. 

79KEYLESS = "keyless" 

80 

81#: Entries before the least-recently-used one is dropped. A resource-safety 

82#: bound on process memory, not a tuning knob: it is sized so that every provider 

83#: a deployment plausibly serves, times the tenants plausibly active at once, 

84#: fits — and so that nothing can grow without limit if that assumption is wrong. 

85CLIENT_CACHE_MAX_ENTRIES = 256 

86 

87#: Seconds an unused entry is kept. Bounds how long a client for a connection 

88#: that has since been deleted can linger when no invalidation arrives. Long 

89#: enough that a normally-busy tenant never pays a rebuild. 

90CLIENT_CACHE_TTL_S = 900.0

93def credential_fingerprint(secret: str, *, salt: bytes | None = None) -> str: 

94 """A stable, salted, non-reversible stand-in for one credential. 

95 

96 `hmac` rather than a bare `sha256(salt + secret)`: it is the construction 

97 designed for keyed digests, and it is constant-time to compare. 

98 """ 

99 if not secret: 

100 return KEYLESS 

101 return hmac.new(salt or _CACHE_KEY_SALT, secret.encode("utf-8"), hashlib.sha256).hexdigest()

126class ClientCache: 

127 """A bounded, expiring, thread-safe LRU of built SDK clients.""" 

128 

129 def __init__( 

130 self, 

131 *, 

132 max_entries: int = CLIENT_CACHE_MAX_ENTRIES, 

133 ttl_s: float = CLIENT_CACHE_TTL_S, 

134 time_func: Callable[[], float] | None = None, 

135 ) -> None: 

136 self._max_entries = max_entries 

137 self._ttl_s = ttl_s 

138 # Monotonic so a wall-clock step can neither expire nor freeze the cache. 

139 self._time = time_func or time.monotonic 

140 self._lock = threading.RLock() 

141 self._entries: OrderedDict[ClientKey, tuple[float, object]] = OrderedDict() 

142 

143 def __len__(self) -> int: 

144 with self._lock: 

145 return len(self._entries) 

146 

147 def get_or_create(self, key: ClientKey, factory: Callable[[], _T]) -> _T: 

148 """The client for `key`, building it once if it is absent or expired. 

149 

150 A factory that RAISES is not cached. A malformed credential must fail 

151 every time rather than once, and a later good build must not be skipped 

152 because an earlier one blew up. The build runs outside the lock, so a slow 

153 construction never blocks another provider's lookup; the double-check on 

154 re-entry keeps the "same credential, same client" guarantee under a race. 

155 """ 

156 now = self._time() 

157 with self._lock: 

158 hit = self._entries.get(key) 

159 if hit is not None and now - hit[0] <= self._ttl_s: 

160 self._entries.move_to_end(key) 

161 return hit[1] # type: ignore[return-value] 

162 if hit is not None: 

163 del self._entries[key] 

164 

165 client = factory() 

166 

167 with self._lock: 

168 existing = self._entries.get(key) 

169 if existing is not None and self._time() - existing[0] <= self._ttl_s: 

170 # A concurrent caller won the race. Return THEIR client so the 

171 # "same credential, one client" property holds even under 

172 # contention; ours is dropped unused. 

173 self._entries.move_to_end(key) 

174 return existing[1] # type: ignore[return-value] 

175 self._entries[key] = (self._time(), client) 

176 self._entries.move_to_end(key) 

177 while len(self._entries) > self._max_entries: 

178 self._entries.popitem(last=False) 

179 return client 

180 

181 def evict_provider(self, provider: str) -> int: 

182 """Drop every entry for one upstream. Returns how many went.""" 

183 with self._lock: 

184 doomed = [key for key in self._entries if key.provider == provider] 

185 for key in doomed: 

186 del self._entries[key] 

187 return len(doomed) 

188 

189 def evict_all(self, reason: str = "") -> int: 

190 """Drop everything. Returns how many entries went.""" 

191 with self._lock: 

192 dropped = len(self._entries) 

193 self._entries.clear() 

194 if dropped: 

195 logger.info("text.client_cache.evicted", entries=dropped, reason=reason) 

196 return dropped
