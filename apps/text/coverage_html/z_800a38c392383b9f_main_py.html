40def _build_observability_config(settings: Settings) -> ObservabilityConfig: 

41 """Carry Text's existing settings into the shared `hope_obs` contract (TASK-987). 

42 

43 `ObservabilityConfig.from_env("text")` alone would miss 

44 `TEXT_OTEL_EXPORTER_ENDPOINT` — Text's own env var name, already resolved 

45 onto `settings.otel_exporter_endpoint` — and the constant identity fields 

46 `Settings` already derives (`otel_service_namespace`, 

47 `otel_deployment_environment`). Narrow those in rather than re-deriving 

48 them from the process environment a second time; `OTEL_SERVICE_NAME` still 

49 overrides `service_name` via `from_env` itself. 

50 """ 

51 base = ObservabilityConfig.from_env("text", service_version="2.0.0") 

52 return replace( 

53 base, 

54 # `or base.otlp_endpoint`, never `or None` (TASK-987 F-22). R-2 makes the 

55 # canonical `OTEL_EXPORTER_OTLP_ENDPOINT` the enable signal, and 

56 # `from_env` has already resolved it. Overwriting with `None` when the 

57 # TEXT_-prefixed name happens to be unset would discard it and take this 

58 # service dark on a manifest that sets only the canonical name. 

59 otlp_endpoint=settings.otel_exporter_endpoint.strip() or base.otlp_endpoint, 

60 service_namespace=settings.otel_service_namespace, 

61 deployment_environment=settings.otel_deployment_environment, 

62 log_level=settings.log_level, 

63 )

66def _register_provider_factories( 

67 registry: ProviderRegistry, http_client: httpx.AsyncClient 

68) -> None: 

69 """Register LAZY provider factories — one per adapter, unconditionally. 

70 

71 Availability used to be gated on "is this provider's ``base_url`` non-empty", 

72 which meant an ENV VAR decided which providers existed. That is backwards in 

73 two ways at once: it made a pure-BYOK tenant's provider 404 because the 

74 PLATFORM had no endpoint (fixed for the cloud four, still true for the local 

75 engines), and it made a provider's existence un-reconfigurable without a 

76 redeploy. 

77 

78 A provider is available iff a CONNECTION resolves for it — and since Text is 

79 stateless, that resolution happens per request, not at boot. So registration 

80 is unconditional and resolution is fail-closed at call time 

81 (`core/connection.py`): a request naming a provider with no resolved 

82 connection gets a typed 503 that names the missing `AiProviderConnection` 

83 row, instead of a 404 that says the provider does not exist. 

84 

85 The (network/SDK-bearing) instance is still built on the first request that 

86 selects it (``registry.get``), never at startup. 

87 """

89 def _shared(builder: Callable[[], LLMProvider]) -> Callable[[], LLMProvider]: 

90 """Memoize a builder so aliased keys resolve to ONE shared instance.""" 

91 box: dict[str, LLMProvider] = {} 

92 

93 def factory() -> LLMProvider: 

94 if "v" not in box: 

95 box["v"] = builder() 

96 return box["v"] 

97 

98 return factory

100 def _register(keys: tuple[str, ...], factory: Callable[[], LLMProvider]) -> None: 

101 for key in keys: 

102 if key not in registry.list_providers(): 

103 registry.register_factory(key, factory)

117 # generic OpenAI-wire one. These two keys used to share ONE instance built 

118 # with the DEFAULT `provider_name`, so a request that said `lm-studio` was
